/*
|--------------------------------------------------------------------------
| SCORE → STATUS
|--------------------------------------------------------------------------
*/

export function statusForScore(
  score: number,
): SkinConcern['status'] {
  if (score >= 75) {
    return 'Excellent';
  }

  if (score >= 55) {
    return 'Good';
  }

  if (score >= 35) {
    return 'Watch';
  }

  return 'Priority';
}

  fallback = 0,
) {

  return Number.isFinite(n)
    ? Math.round(n)
    : fallback;
}

# Replit commands

After pasting the two files, restart the Expo app.

If Replit has a terminal, run:

```bash
npm install
```

Then:

```bash
npx expo start
```

If your project already has its normal Replit start command, simply restart that command instead.

You **do not need to install another skincare library** for this implementation.

---

# One important thing to check

Your existing project must already have this route registered in:

```text
app/(tabs)/_layout.tsx
```

It should have something equivalent to:

```tsx
<Tabs.Screen
  name="skin-care"
  options={{
    title: 'Skin',
  }}
/>
```

If the **Skin tab itself is missing**, the two files above won't make it appear in navigation. The screen file exists, but Expo Router still needs the tab registered.

Also make sure the file name is **exactly**:

```text
skin-care.tsx
```

inside:

```text
app/(tabs)/
```

not:

```text
app/skin-care.tsx
```

and not:

```text
app/(tabs)/skin.tsx
```
