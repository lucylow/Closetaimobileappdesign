'use client'

import { useState } from 'react'
import Image from 'next/image'
import { motion, AnimatePresence } from 'framer-motion'
import { Plus, Camera, Grid, X, Heart } from 'lucide-react'
import { Button } from '@/components/ui/button'
import { Badge } from '@/components/ui/badge'
import { Card, CardContent } from '@/components/ui/card'

const mockWardrobe = [
  { id: 1, category: 'tops', src: '/api/placeholder/300/400', tags: ['overshirt', 'linen'], liked: true },
  { id: 2, category: 'shoes', src: '/api/placeholder/300/400', tags: ['sneakers', 'white'], liked: false },
  { id: 3, category: 'bags', src: '/api/placeholder/300/400', tags: ['tote', 'leather'], liked: true },
  { id: 4, category: 'tops', src: '/api/placeholder/300/400', tags: ['t-shirt', 'cotton'], liked: false },
  { id: 5, category: 'bottoms', src: '/api/placeholder/300/400', tags: ['jeans', 'slim'], liked: true },
  { id: 6, category: 'jewelry', src: '/api/placeholder/300/400', tags: ['necklace', 'gold'], liked: false }
]

export default function WardrobePage() {
  const [selectedCategory, setSelectedCategory] = useState('all')
  const [uploadOpen, setUploadOpen] = useState(false)
  const [layout, setLayout] = useState<'grid' | 'masonry'>('grid')

  const categories = ['all', 'tops', 'shoes', 'bags', 'bottoms', 'jewelry'] as const

  return (
    <div className="min-h-dvh p-4 md:p-6 lg:p-8 space-y-6">
      {/* Mobile Header */}
      <div className="flex flex-col lg:flex-row lg:items-center justify-between gap-4">
        <div>
          <h1 className="text-3xl md:text-4xl font-black bg-gradient-to-r from-slate-900 to-indigo-900 bg-clip-text text-transparent">
            My Wardrobe
          </h1>
          <p className="text-xl text-slate-600 mt-2">156 items • 12 categories</p>
        </div>
        <div className="flex items-center gap-2">
          <Button
            variant="outline"
            size="sm"
            onClick={() => setLayout(layout === 'grid' ? 'masonry' : 'grid')}
            className="h-12 w-12 p-0"
          >
            {layout === 'grid' ? <Grid className="h-5 w-5" /> : <View className="h-5 w-5" />}
          </Button>
          <Button
            onClick={() => setUploadOpen(true)}
            className="h-14 px-6 shadow-xl bg-gradient-to-r from-indigo-600 to-purple-600 hover:from-indigo-700 hover:to-purple-700"
          >
            <Camera className="h-5 w-5 mr-2" />
            Upload
          </Button>
        </div>
      </div>

      {/* Mobile Category Pills */}
      <div className="flex flex-wrap gap-2">
        {categories.map(cat => (
          <Button
            key={cat}
            variant={selectedCategory === cat ? 'default' : 'outline'}
            size="sm"
            className={`h-12 px-4 rounded-full ${selectedCategory === cat ? 'bg-indigo-600 hover:bg-indigo-700 shadow-md' : ''}`}
            onClick={() => setSelectedCategory(cat)}
          >
            {cat}
          </Button>
        ))}
      </div>

      {/* Mobile Masonry/Grid Gallery */}
      <div className={`grid gap-3 ${layout === 'grid' ? 'grid-cols-2 lg:grid-cols-3 xl:grid-cols-4' : 'grid-cols-1 md:grid-cols-2 lg:grid-cols-3'}`}>
        {mockWardrobe.map((item, index) => (
          <motion.div
            key={item.id}
            initial={{ opacity: 0, scale: 0.9 }}
            whileInView={{ opacity: 1, scale: 1 }}
            viewport={{ once: true }}
            whileHover={{ scale: 1.02 }}
            transition={{ delay: index * 0.05 }}
          >
            <Card className="group relative overflow-hidden border-0 shadow-lg hover:shadow-2xl hover:-translate-y-2 transition-all duration-300 bg-white/80 backdrop-blur-xl h-full">
              <div className="relative h-64 lg:h-72">
                <Image
                  src={item.src}
                  alt={item.category}
                  fill
                  className="object-cover group-hover:scale-110 transition-transform duration-500"
                />
                <div className="absolute inset-0 bg-gradient-to-t from-black/20 to-transparent opacity-0 group-hover:opacity-100 transition-opacity duration-300" />
                
                {/* Like Button */}
                <motion.button
                  className="absolute top-4 right-4 p-2 bg-white/90 backdrop-blur-sm rounded-2xl shadow-lg hover:shadow-xl transition-all duration-200"
                  whileTap={{ scale: 0.95 }}
                  whileHover={{ scale: 1.1 }}
                >
                  <Heart className={`h-5 w-5 transition-colors ${item.liked ? 'text-red-500 fill-red-500' : 'text-slate-400 hover:text-red-500'}`} />
                </motion.button>

                {/* Category Badge */}
                <Badge className="absolute bottom-4 left-4 bg-white/90 backdrop-blur-sm text-slate-800 font-bold">
                  {item.category}
                </Badge>
              </div>
              
              <CardContent className="p-4 pt-0">
                <div className="flex flex-wrap gap-1 mt-2">
                  {item.tags.slice(0, 2).map(tag => (
                    <Badge key={tag} variant="secondary" className="text-xs px-2 py-1">
                      #{tag}
                    </Badge>
                  ))}
                </div>
              </CardContent>
            </Card>
          </motion.div>
        ))}
      </div>

      {/* Mobile FAB */}
      <motion.div
        className="fixed bottom-6 right-6 lg:hidden z-50"
        whileTap={{ scale: 0.95 }}
        initial={{ scale: 0 }}
        animate={{ scale: 1 }}
      >
        <Button
          size="lg"
          className="h-16 w-16 p-0 shadow-2xl bg-gradient-to-r from-indigo-600 to-purple-600 hover:from-indigo-700 hover:to-purple-700 shadow-indigo-500/25"

        >
          <Plus className="h-6 w-6" />

            onClick={() => setUploadOpen(false)}
          >
            <motion.div
              initial={{ scale: 0.9, opacity: 0 }}

              exit={{ scale: 0.9, opacity: 0 }}
              className="bg-white/95 backdrop-blur-xl rounded-3xl p-8 max-w-sm w-full max-h-[90vh] overflow-y-auto"
              onClick={e => e.stopPropagation()}
            >
              <div className="text-center space-y-6">
                <div className="w-20 h-20 mx-auto bg-gradient-to-r from-indigo-500 to-purple-600 rounded-3xl flex items-center justify-center shadow-2xl">
                  <Camera className="h-10 w-10 text-white" />
                </div>
                <div>
                  <h2 className="text-2xl font-bold text-slate-900 mb-2">Upload New Item</h2>
                  <p className="text-slate-600">Drag & drop or tap to browse (JPG, PNG < 50MB)</p>
