-- src/backend/sql/init.sql
-- COMPLETE ClosetAI PostgreSQL Schema for Replit

-- Users & Authentication
CREATE TABLE users (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  email VARCHAR(255) UNIQUE NOT NULL,
  display_name VARCHAR(100) NOT NULL,
  is_guest BOOLEAN DEFAULT FALSE,
  style_profile JSONB DEFAULT '{}',
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW()
);

-- Wardrobe Items (Core Data Model)
CREATE TABLE wardrobe_items (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES users(id) ON DELETE CASCADE,
  name VARCHAR(200) NOT NULL,
  category VARCHAR(50) CHECK (category IN ('top','bottom','dress','outerwear','shoes','accessory')),
  color VARCHAR(50),
  brand VARCHAR(100),
  image_key TEXT NOT NULL, -- Replit Object Storage path
  ai_tags TEXT[], -- ['casual','linen','breathable']
  usage_count INTEGER DEFAULT 0,
  last_worn TIMESTAMP,
  metadata JSONB, -- Vision AI results
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW()
);
CREATE INDEX idx_wardrobe_user ON wardrobe_items(user_id);
CREATE INDEX idx_wardrobe_category ON wardrobe_items(category);

-- Outfits (AI Generated)
CREATE TABLE outfits (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES users(id) ON DELETE CASCADE,
  name VARCHAR(200),
  item_ids UUID[] NOT NULL, -- Array of wardrobe_item.ids
  occasion VARCHAR(50),
  weather VARCHAR(50),
  ai_score NUMERIC(3,2), -- 0.00-1.00
  explanation TEXT,
  tags TEXT[],
  preview_image TEXT, -- Composite thumbnail
  created_at TIMESTAMP DEFAULT NOW()
);
CREATE INDEX idx_outfits_user ON outfits(user_id);

-- Try-On Jobs (Async Processing)
CREATE TABLE tryon_jobs (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES users(id) ON DELETE CASCADE,
  selfie_image_key TEXT NOT NULL,
  garment_ids UUID[] NOT NULL,
  status VARCHAR(20) DEFAULT 'queued' CHECK (status IN ('queued','processing','completed','failed')),
  result_image_key TEXT,
  confidence NUMERIC(3,2),
  error_message TEXT,
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW()
);
CREATE INDEX idx_tryon_status ON tryon_jobs(status);
CREATE INDEX idx_tryon_user ON tryon_jobs(user_id);

-- Content Generation Jobs
CREATE TABLE content_jobs (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES users(id) ON DELETE CASCADE,
  outfit_id UUID REFERENCES outfits(id),
  tone VARCHAR(50),
  platform VARCHAR(20),
  status VARCHAR(20) DEFAULT 'queued',
  caption TEXT,
  hashtags TEXT[],
  image_key TEXT,
  created_at TIMESTAMP DEFAULT NOW()
);

-- Usage Credits (Freemium Model)
CREATE TABLE user_credits (
  user_id UUID PRIMARY KEY REFERENCES users(id) ON DELETE CASCADE,
  credits INTEGER DEFAULT 100,
  used_today INTEGER DEFAULT 0,
  updated_at TIMESTAMP DEFAULT NOW()
);

-- User Feedback (ML Training)
CREATE TABLE feedback (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES users(id),
  outfit_id UUID REFERENCES outfits(id),
  rating INTEGER CHECK (rating >= -1 AND rating <= 1),
  comment TEXT,
  created_at TIMESTAMP DEFAULT NOW()
);

-- Analytics Events
CREATE TABLE events (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES users(id),
  event_type VARCHAR(50),
  payload JSONB,
  created_at TIMESTAMP DEFAULT NOW()
);
CREATE INDEX idx_events_user_type ON events(user_id, event_type);

-- Image Metadata (Vision AI)
CREATE TABLE image_metadata (
  image_key TEXT PRIMARY KEY,
  wardrobe_item_id UUID REFERENCES wardrobe_items(id),
  colors JSONB, -- Dominant color palette
  objects JSONB, -- Detected clothing types
  processed_at TIMESTAMP DEFAULT NOW()
);

-- Sessions (JWT Tokens)
CREATE TABLE sessions (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES users(id) ON DELETE CASCADE,
  token_hash TEXT NOT NULL,
  expires_at TIMESTAMP NOT NULL,
  created_at TIMESTAMP DEFAULT NOW()
);
CREATE INDEX idx_sessions_token ON sessions(token_hash);
CREATE INDEX idx_sessions_expires ON sessions(expires_at);

Auth Routes (JWT + Guest Mode)
typescript
// backend/src/routes/auth.ts
import { Router, Request, Response } from 'express';
import { Pool } from 'pg';
import jwt from 'jsonwebtoken';

const router = Router();
const pool = new Pool({ connectionString: process.env.DATABASE_URL! });

router.post('/guest', async (req: Request, res: Response) => {
  const { display_name = 'Guest' } = req.body;
  
  const client = await pool.connect();
  try {
    // Create or get guest user
    const userResult = await client.query(
      `INSERT INTO users (email, display_name, is_guest) 
       VALUES ($1, $2, true) 
       ON CONFLICT (email) DO UPDATE SET display_name = $2 
       RETURNING id, display_name, is_guest`,
      [`guest_${Date.now()}@closetai.com`, display_name]
    );
    
    const user = userResult.rows[0];
    const token = jwt.sign({ userId: user.id }, process.env.JWT_SECRET!, { expiresIn: '7d' });
    
    res.json({ token, user });
  } finally {
    client.release();
  }
});

export default router;

async function getUserWardrobe(req: any, res: any) {
  const userId = req.user.userId;
  
  const result = await pool.query(
    `SELECT * FROM wardrobe_items WHERE user_id = $1 ORDER BY created_at DESC`,
    [userId]
  );
  
  res.json(result.rows);
}

GlassCard Component (Core UI Primitive)

// frontend/src/components/common/GlassCard.tsx
import React from 'react';
import { View, Text, StyleSheet, Pressable } from 'react-native';
import Animated, { useSharedValue, useAnimatedStyle, withSpring } from 'react-native-reanimated';

interface GlassCardProps {
  title: string;
  subtitle?: string;
  children: React.ReactNode;
  onPress?: () => void;
}

export const GlassCard: React.FC<GlassCardProps> = ({
  title, subtitle, children, onPress
}) => {
  const scale = useSharedValue(1);

  const animatedStyle = useAnimatedStyle(() => ({
    transform: [{ scale: scale.value }],
  }));

  const handlePressIn = () => {
    scale.value = withSpring(0.97);
  };

  const handlePressOut = () => {
    scale.value = withSpring(1);
  };

  return (
    <Animated.View style={[styles.card, animatedStyle]}>
      <Pressable 
        style={styles.content}
        onPress={onPress}
        onPressIn={handlePressIn}
        onPressOut={handlePressOut}
        android_ripple={{ color: 'rgba(255,255,255,0.2)' }}
      >
        <Text style={styles.title}>{title}</Text>
        {subtitle && <Text style={styles.subtitle}>{subtitle}</Text>}
        {children}
      </Pressable>
    </Animated.View>
  );
};

const styles = StyleSheet.create({
  card: {
    backgroundColor: 'rgba(255, 255, 255, 0.25)',
    backdropFilter: 'blur(20px)',
    borderRadius: 24,
    padding: 24,

// frontend/src/screens/WardrobeScreen.tsx
import React, { useEffect, useState } from 'react';
import { FlatList, View, StyleSheet } from 'react-native';
import { ScreenContainer } from '../components/common/ScreenContainer';
import { WardrobeItemCard } from '../components/wardrobe/WardrobeItemCard';
import { useWardrobe } from '../hooks/useWardrobe';
import { PrimaryButton } from '../components/common/PrimaryButton';

export const WardrobeScreen: React.FC = () => {
  const { wardrobe, loading, refreshWardrobe } = useWardrobe();
  const [filter, setFilter] = useState<'all' | 'top' | 'bottom' | 'dress'>('all');

  const filteredWardrobe = wardrobe.filter(item => 
    filter === 'all' || item.category === filter
  );

  const renderItem = ({ item }: { item: any }) => (
    <WardrobeItemCard item={item} onPress={() => {}} />
  );

  return (
    <ScreenContainer>
      <FlatList
        data={filteredWardrobe}
        renderItem={renderItem}
        keyExtractor={(item) => item.id}
        numColumns={2}
        columnWrapperStyle={styles.row}
        contentContainerStyle={styles.list}
        onRefresh={refreshWardrobe}
        refreshing={loading}
        ListHeaderComponent={
          <View style={styles.filters}>
            {(['all', 'top', 'bottom', 'dress'] as const).map(cat => (
              <Pressable
                key={cat}
                style={[
                  styles.filterChip,
                  filter === cat && styles.filterChipActive
                ]}
                onPress={() => setFilter(cat)}
              >
                <Text style={styles.filterText}>{cat.toUpperCase()}</Text>
              </Pressable>
            ))}
          </View>
        }
        ListFooterComponent={
          <PrimaryButton 
            title="+ Add New Item" 
            onPress={() => navigation.navigate('AddItem')}
          />
        }
      />
    </ScreenContainer>
  );
};
