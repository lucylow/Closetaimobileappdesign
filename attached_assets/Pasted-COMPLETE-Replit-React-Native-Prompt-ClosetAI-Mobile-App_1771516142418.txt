3 Upload Patterns (Choose Best):
Pattern 1: Client → Backend → Storage (Recommended)
text
1. Mobile: POST /api/storage/upload-init {filename, contentType}
2. Backend: Generate key → Return signed PUT URL  
3. Mobile: PUT image to signed URL (direct to storage)
4. Backend: Save metadata to PostgreSQL

Pattern 2: Multipart via Backend
text
1. Mobile: POST /api/wardrobe with FormData (image)
2. Backend: Receive buffer → Upload to Object Storage
3. Backend: Save PostgreSQL record

Pattern 3: Base64 (Small Images Only)
text
1. Mobile: POST /api/wardrobe {imageBase64, metadata}
2. Backend: Buffer.from(base64) → Object Storage

-- ClosetAI Object Storage Schema

-- Storage Objects (All Images)
CREATE TABLE storage_objects (
  key TEXT PRIMARY KEY,
  bucket TEXT NOT NULL DEFAULT 'closetai-media',
  user_id UUID,
  content_type TEXT NOT NULL,
  size_bytes BIGINT NOT NULL,
  metadata JSONB,
  uploaded_at TIMESTAMP DEFAULT NOW(),
  expires_at TIMESTAMP, -- TTL for temp files
  is_public BOOLEAN DEFAULT FALSE
);
CREATE INDEX idx_storage_user ON storage_objects(user_id);
CREATE INDEX idx_storage_expires ON storage_objects(expires_at);

-- Wardrobe Items (Links to Storage)
CREATE TABLE wardrobe_items (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL REFERENCES users(id),
  name VARCHAR(200) NOT NULL,
  category VARCHAR(50),
  image_key TEXT NOT NULL REFERENCES storage_objects(key) ON DELETE CASCADE,
  thumbnail_key TEXT REFERENCES storage_objects(key),
  ai_tags TEXT[],
  usage_count INTEGER DEFAULT 0,
  created_at TIMESTAMP DEFAULT NOW()
);

-- Try-On Jobs (Input + Output Images)
CREATE TABLE tryon_jobs (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL,
  selfie_key TEXT NOT NULL REFERENCES storage_objects(key),
  garment_keys TEXT[] NOT NULL, -- Array of wardrobe image_keys
  result_key TEXT REFERENCES storage_objects(key), -- Generated try-on
  status VARCHAR(20) DEFAULT 'queued',
  created_at TIMESTAMP DEFAULT NOW()
);

-- Outfit Composites (AI Generated Previews)
CREATE TABLE outfit_previews (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL,
  outfit_items TEXT[], -- wardrobe_item keys
  composite_key TEXT NOT NULL REFERENCES storage_objects(key),
  created_at TIMESTAMP DEFAULT NOW()
);


💾 REPLIT OBJECT STORAGE CLIENT (Complete Implementation)
typescript
// backend/src/storage/replitStorage.ts
import { ReplitStorage } from '@replit/storage'; // Hypothetical SDK
// Fallback to fetch-based S3-compatible API

class ClosetAIStorage {
  private bucket: string;
  private client: any;

  constructor() {
    this.bucket = process.env.REPLIT_STORAGE_BUCKET || 'closetai-media';
    // Initialize Replit Storage client
  }

  // Generate deterministic storage keys
  userKey(userId: string, type: string, id: string, ext: string): string {
    return `users/${userId}/${type}/${id}.${ext}`;
  }

  wardrobeItemKey(userId: string, itemId: string): string {
    return this.userKey(userId, 'wardrobe', itemId, 'jpg');
  }

  tryOnResultKey(userId: string, jobId: string): string {
    return this.userKey(userId, 'tryon', jobId, 'png');
  }

  // Upload buffer directly to Object Storage
  async uploadBuffer(
    buffer: Buffer, 
    key: string, 
    contentType: string = 'image/jpeg'
  ): Promise<string> {
    const result = await this.client.put(key, buffer, {
      contentType,
      metadata: { uploadedBy: 'closetai-api' }
    });
    return key;
  }

  // Generate signed URLs (15min expiry)
  async getSignedUrl(key: string, operation: 'get' | 'put' = 'get', expiresIn = 900): Promise<string> {
    return this.client.signedUrl(key, operation, expiresIn);
  }

  // Multi-file upload (wardrobe batch)
  async uploadWardrobeBatch(userId: string, files: Array<{buffer: Buffer, itemId: string}>): Promise<string[]> {
    const keys = files.map(f => this.wardrobeItemKey(userId, f.itemId));
    const results = await Promise.all(
      files.map((f, i) => this.uploadBuffer(f.buffer, keys[i]))
    );
    return results;
  }

  // Cleanup expired objects (cron job)
  async cleanupExpired(): Promise<number> {
    const expired = await this.client.list({ prefix: '', maxKeys: 1000 });
    const deleted = await Promise.all(
      expired.filter(obj => obj.expiresAt && new Date(obj.expiresAt) < new Date())
             .map(obj => this.client.delete(obj.key))
    );
    return deleted.length;
  }
}

export const storage = new ClosetAIStorage();

    setUploading(true);
    setProgress(0);

      // 1. Get signed upload URL

        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          userId,
          filename: `wardrobe-${Date.now()}.jpg`,
          contentType: 'image/jpeg'
        })
      });

      const { key, signedUrl } = await response.json();

      // 2. Upload directly to Object Storage

        uri,
        type: 'image/jpeg',
        name: key,

import { Router, Request, Response } from 'express';
import { storage } from '../storage/replitStorage';
import { authenticateToken } from '../middleware/auth';

const router = Router();

// Get signed URL for direct client upload
router.post('/upload-init', async (req: Request, res: Response) => {
  const { userId, filename, contentType } = req.body;

  const signedUrl = await storage.getSignedUrl(key, 'put', 900); // 15min
  
  res.json({
    key,
    signedUrl,
    publicUrl: key // After upload completes
  });
});

// List user's images

    `SELECT s.key, s.content_type, s.size_bytes, wi.name 
     FROM storage_objects s
     LEFT JOIN wardrobe_items wi ON s.key = wi.image_key
     WHERE s.user_id = $1`,
    [userId]
  );
  
  // Generate signed view URLs

    images.rows.map(async (img: any) => ({
      ...img,

  );
  
  res.json(signedImages);
});

SignedImage Component (Auto-Expires)
tsx
// frontend/src/components/storage/SignedImage.tsx
import React, { useEffect, useState } from 'react';
import { Image, ImageSourcePropType, StyleProp, ViewStyle, ActivityIndicator } from 'react-native';

interface SignedImageProps {
  imageKey: string;
  style?: StyleProp<ViewStyle>;
  resizeMode?: 'cover' | 'contain';
}

export const SignedImage: React.FC<SignedImageProps> = ({
  imageKey,
  style,
  resizeMode = 'cover'
}) => {
  const [signedUrl, setSignedUrl] = useState<string | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    fetchSignedUrl();
  }, [imageKey]);

  const fetchSignedUrl = async () => {
    try {
      const response = await fetch(`/api/storage/${imageKey}/signed`);
      const { url } = await response.json();
      setSignedUrl(url);
    } catch (error) {
      console.error('Failed to get signed URL:', error);
    } finally {
      setLoading(false);
    }
  };

  if (loading) {
    return <ActivityIndicator style={style} color="#6E4AE0" />;
  }

  if (!signedUrl) {
    return <View style={style} />;
  }

  return (
    <Image
      source={{ uri: signedUrl }}
      style={style}
      resizeMode={resizeMode}
      onError={() => fetchSignedUrl()} // Auto-refresh expired URLs
    />
  );
};
